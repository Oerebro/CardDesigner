$folderPath = "C:\Users\heile\Documents\Programming\CardDesigner\resources\img\armor\pants" 

Get-ChildItem -Path $folderPath -File -Recurse | ForEach-Object {
    $originalName = $_.Name
    $fullPath = $_.FullName
    $dir = $_.DirectoryName

    # Normalize extension (lowercase)
    $ext = $_.Extension.ToLower()

    # Base name without extension
    $baseName = [System.IO.Path]::GetFileNameWithoutExtension($originalName)

    # Remove double extensions like .png.png or .webp.png
    if ($baseName.ToLower().EndsWith(".png") -or $baseName.ToLower().EndsWith(".webp")) {
        $baseName = [System.IO.Path]::GetFileNameWithoutExtension($baseName)
    }

    # Always rename to .png
    $newName = "$baseName.png"
    $newPath = Join-Path $dir $newName

    if ($originalName -ne $newName) {
        if (-not (Test-Path $newPath)) {
            Rename-Item -Path $fullPath -NewName $newName
            Write-Host "Renamed: $originalName → $newName"
        } else {
            Write-Host "Skipping (target exists): $newName"
        }
    }
}
